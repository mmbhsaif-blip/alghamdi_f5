محلي (وضع التجربة)
طريقة إعداد سحابة جوجل في دقيقة:
  1. افتحي حساب Google Drive الخاص بالمدرسة وقومي بإنشاء **جدول بيانات جديد (Google Sheet)**.
  2. من القائمة العلوية للجدول، اضغطي على **"ملحقات (Extensions)"** ثم **"Apps Script"**.
  3. احذفي الكود المكتوب هناك، والصقي الكود المخصص بالبطاقة أدناه، ثم اضغطي **"حفظ 💾"**.
  4. اضغطي على زر **"نشر (Deploy)"** باللون الأزرق بالأعلى، ثم **"نشر جديد (New deployment)"**.
  5. حددي النوع كـ **"تطبيق ويب (Web app)"**، واجعلي صلاحية الدخول لـ **"أي شخص (Anyone)"**، ثم اضغطي Deploy.
  6. انسخي رابط الويب الممنوح لكِ، وألصقيه في الحقل المخصص هنا واضغطي حفظ!
📋 الكود البرمجي لنسخه ولصقه في Apps Script (المطور والمحصن ذاتياً للرابط المستقل والمربوط): 
/* كود محرك المزامنة السحابي المطور والذكي لجداول بيانات جوجل */
// ⚠️ تم تضمين رابط جدول البيانات الجديد الخاص بمدرستكم تلقائياً أدناه للربط المستقر والمباشر:
var SPREADSHEET_URL = "https://docs.google.com/spreadsheets/d/1GBtbYZ4xDaeqvT2H7jFtkn6LfWPKMVmAiF1P5WUBIZE/edit?gid=0#gid=0"; 

function getSpreadsheet() {
  if (SPREADSHEET_URL && SPREADSHEET_URL.trim() !== "") {
    try {
      return SpreadsheetApp.openByUrl(SPREADSHEET_URL.trim());
    } catch(e) {
      return SpreadsheetApp.getActive();
    }
  }
  return SpreadsheetApp.getActive();
}

function doGet(e) {
  try {
    var op = e.parameter.action;
    var ss = getSpreadsheet();
    if (!ss) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: "لم نجد جدول البيانات النشط. يرجى مراجعة رابط ملف الإكسل." }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var staffSheet = ss.getSheetByName("Staff") || ss.insertSheet("Staff");
    var configSheet = ss.getSheetByName("Config") || ss.insertSheet("Config");
    var attSheet = ss.getSheetByName("Attendance") || ss.insertSheet("Attendance");
    
    if (staffSheet.getLastRow() === 0) staffSheet.appendRow(["ID", "Name", "Type", "Token"]);
    if (configSheet.getLastRow() === 0) configSheet.appendRow(["Key", "Value"]);
    if (attSheet.getLastRow() === 0) attSheet.appendRow(["ID", "Name", "Date", "Time", "Action", "Method", "Status", "Reason"]);
    
    if (op === "getData") {
      var staffData = staffSheet.getDataRange().getValues();
      var staff = {};
      for (var i = 1; i < staffData.length; i++) {
        staff[staffData[i][0]] = { id: String(staffData[i][0]), name: staffData[i][1], type: staffData[i][2], token: staffData[i][3] || null };
      }
      var configData = configSheet.getDataRange().getValues();
      var config = {};
      for (var i = 1; i < configData.length; i++) {
        config[configData[i][0]] = configData[i][1];
      }
      var attData = attSheet.getDataRange().getValues();
      var attendance = [];
      for (var i = attData.length - 1; i >= 1; i--) {
        attendance.push({ id: String(attData[i][0]), name: attData[i][1], date: attData[i][2], time: attData[i][3], action: attData[i][4], method: attData[i][5], status: attData[i][6], reason: attData[i][7] });
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true, staff: staff, config: config, attendance: attendance }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (op === "addAttendance") {
      var attSheet = ss.getSheetByName("Attendance") || ss.insertSheet("Attendance");
      attSheet.appendRow([e.parameter.id, e.parameter.name, e.parameter.date, e.parameter.time, e.parameter.actionType, e.parameter.method, e.parameter.status, e.parameter.reason]);
      return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (op === "addStaff") {
      var staffSheet = ss.getSheetByName("Staff") || ss.insertSheet("Staff");
      staffSheet.appendRow([e.parameter.id, e.parameter.name, e.parameter.type, e.parameter.token || ""]);
      return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (op === "updateToken") {
      var staffSheet = ss.getSheetByName("Staff");
      var staffData = staffSheet.getDataRange().getValues();
      for (var i = 1; i < staffData.length; i++) {
        if (String(staffData[i][0]) === String(e.parameter.id)) {
          staffSheet.getRange(i + 1, 4).setValue(e.parameter.token || "");
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (op === "deleteStaff") {
      var staffSheet = ss.getSheetByName("Staff");
      var staffData = staffSheet.getDataRange().getValues();
      for (var i = 1; i < staffData.length; i++) {
        if (String(staffData[i][0]) === String(e.parameter.id)) {
          staffSheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (op === "deleteAttendance") {
      var attSheet = ss.getSheetByName("Attendance");
      var attData = attSheet.getDataRange().getValues();
      for (var i = 1; i < attData.length; i++) {
        if (String(attData[i][0]) === String(e.parameter.id) && String(attData[i][2]) === String(e.parameter.date) && String(attData[i][3]) === String(e.parameter.time) && String(attData[i][4]) === String(e.parameter.actionType)) {
          attSheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (op === "resetSystem") {
      var sheets = ss.getSheets();
      for (var i = 0; i < sheets.length; i++) {
        if (sheets[i].getName() !== "Temp") {
          try { ss.deleteSheet(sheets[i]); } catch(err) {}
        }
      }
      ss.insertSheet("Staff").appendRow(["ID", "Name", "Type", "Token"]);
      ss.insertSheet("Attendance").appendRow(["ID", "Name", "Date", "Time", "Action", "Method", "Status", "Reason"]);
      ss.insertSheet("Config").appendRow(["Key", "Value"]);
      return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
